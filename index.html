// Lógica do Timer Dinâmico de Artes Marciais let timeLeft = 180; // 3 minutos em segundos let timerId = null; let isRunning = false; let currentRound = 1; const totalRounds = 3; const display = document.getElementById('display'); const startBtn = document.getElementById('startBtn'); const statusRound = document.getElementById('status-round'); function updateDisplay() { const minutes = Math.floor(timeLeft / 60); const seconds = timeLeft % 60; display.textContent = `${minutes.toString().padStart(2, '0')}:${seconds.toString().padStart(2, '0')}`; } function startTimer() { if (isRunning) { clearInterval(timerId); startBtn.textContent = "RETOMAR COMBATE"; startBtn.style.background = "#45f3ff"; startBtn.style.color = "#000"; isRunning = false; } else { isRunning = true; startBtn.textContent = "PAUSAR TREINO"; startBtn.style.background = "#ff0055"; startBtn.style.color = "#fff"; timerId = setInterval(() => { if (timeLeft > 0) { timeLeft--; updateDisplay(); } else { // Fim do round if (currentRound < totalRounds) { currentRound++; timeLeft = 180; // reseta para 3 min statusRound.textContent = `Round ${currentRound} / ${totalRounds}`; alert(`Fim do Round! Prepare-se para o Round ${currentRound}`); } else { clearInterval(timerId); alert("Treino Concluído! Excelente trabalho, Guerreiro."); resetTimer(); } } }, 1000); } } function resetTimer() { timeLeft = 180; currentRound = 1; isRunning = false; startBtn.textContent = "INICIAR COMBATE"; statusRound.textContent = `Round 1 / 3`; updateDisplay(); } startBtn.addEventListener('click', startTimer); updateDisplay();
